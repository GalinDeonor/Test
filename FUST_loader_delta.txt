-- ЛОАДЕР «Пьяный Петух» 0.65 DELTA — ANDROID
-- Скачивает 6 частей (Fust1..Fust6) с GitHub, склеивает и запускает.
-- ПЕРЕД ЗАПУСКОМ: залей в КОРЕНЬ репозитория GalinDeonor/Test ОДНИМ КОММИТОМ:
--   Fust1..Fust6, manifest.json, BUGS.md, FUST_loader_delta.txt
-- ЛОАДЕР FUST_loader_universal.txt НЕ МЕНЯЕТСЯ: сам качает новую версию через manifest.
-- 0.65: ПОЛНАЯ ДИАГНОСТИКА кода (0 багов, чистка, оптимизация).
-- 0.64: АНТИ-АФК (без кика за бездействие), HUD ИНВЕНТАРЯ цели
-- (из VOMAGLA), ХИТЫ: хитмаркер/хитлог/звуки попадания и убийства.
-- 0.63: СМЕНА FOV (поле зрения 30-120, Аимбот -> Экран) — взято из
-- найденного тобой исходника, реализовано БЕЗ хуков (безопасно Delta).
-- 0.62: КНОПКИ БОЛЬШЕ НЕ ТЕРЯЮТСЯ ЗА КРАЕМ ЭКРАНА (кнопка-петух, XRAY,
-- CAM, панель фрикама: клэмп при драге + автоворачивание в экран при
-- повороте телефона). Имена файлов на репо ВЕЧНЫЕ с 0.61 — лоадер лежит
-- как FUST_loader_delta.txt и НЕ МЕНЯЕТ ИМЯ, только содержимое.
-- Внутри всё 0.60-0.61: конфиги без клавиатуры (автонейм «конфиг-1»),
-- X-RAY В ОДНОМ МОЩНОМ тумблере с хоткеем для ПК, «Заморозить HUD»,
-- панель CAM пальцем, «Прозрачность фона меню», мульти-точки фрикама,
-- стрельба, FOV-круг.
print("[ПП 0.65] Лоадер запущен, качаю 6 частей с GitHub...")
local B = "https://raw.githubusercontent.com/GalinDeonor/Test/refs/heads/main/Fust"
local want = {94651, 94651, 94651, 94651, 94651, 94651}
local TOTAL = 567906
local src = ""
for i = 1, 6 do
        local r
        local u = B .. i .. "?v=65"
        local ok, rr = pcall(game.HttpGet, game, u)
        if not ok or type(rr) ~= "string" or #rr < 1000 then
                local rf = request or http_request
                if rf then
                        local ok2, rr2 = pcall(rf, {Url = u, Method = "GET"})
                        if ok2 and type(rr2) == "table" then rr = rr2.Body or rr2.body end
                end
        end
        if type(rr) == "string" and #rr >= want[i] then
                r = rr
                print("[ПП 0.65] Часть " .. i .. ": " .. #rr .. " байт")
        else
                warn("[ПП 0.65] Часть " .. i .. " битая или не залита (нужно " .. want[i] .. " байт). Перезалей Fust" .. i .. " на GitHub.")
                return
        end
        src = src .. r
end
print("[ПП 0.65] Всего: " .. #src .. " байт")
if #src < TOTAL then
        warn("[ПП 0.65] Сумма меньше " .. TOTAL .. " -- какая-то часть битая, скинь скрин консоли.")
        return
end
local fn, err = (loadstring or load)(src, "=Fust")
if not fn then
        warn("[ПП 0.65] Компиляция: " .. tostring(err))
        return
end
local ok, rerr = pcall(fn)
if not ok then
        warn("[ПП 0.65] Выполнение: " .. tostring(rerr))
        warn("[ПП 0.65] Скинь СКРИН консоли -- там будет строка [FUST] с местом падения.")
end
