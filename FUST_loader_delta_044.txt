-- Скачивает 6 частей (Fust1..Fust6) с GitHub, склеивает и запускает.
-- ПЕРЕД ЗАПУСКОМ: залей Fust1..Fust6 из архива 0.44 в КОРЕНЬ репозитория
-- GalinDeonor/Test (Add file -> Upload files -> Commit), старые замени.
-- ФИКС 0.44: меню на телефоне строится через Scale (85%x70%) — открыть
-- его гигантским за экраном физически больше невозможно.
local B = "https://raw.githubusercontent.com/GalinDeonor/Test/refs/heads/main/Fust"
local want = {69895, 69909, 69900, 69946, 69952, 69756}
local TOTAL = 419358
local src = ""
for i = 1, 6 do
        local r
        local u = B .. i .. "?v=44"
        local ok, rr = pcall(game.HttpGet, game, u)
        if not ok or type(rr) ~= "string" or #rr < 1000 then
                local rf = request or http_request
                if rf then
                        local ok2, rr2 = pcall(rf, {Url = u, Method = "GET"})
                        if ok2 and type(rr2) == "table" then rr = rr2.Body or rr2.body end
                end
        end
        if type(rr) == "string" and #rr >= want[i] then
                r = rr
                print("[ПП 0.44] Часть " .. i .. ": " .. #rr .. " байт")
        else
                warn("[ПП 0.44] Часть " .. i .. " битая или не залита (нужно " .. want[i] .. " байт). Перезалей Fust" .. i .. " на GitHub.")
                return
        end
        src = src .. r
end
print("[ПП 0.44] Всего: " .. #src .. " байт")
if #src < TOTAL then
        warn("[ПП 0.44] Сумма меньше " .. TOTAL .. " -- какая-то часть битая, скинь скрин консоли.")
        return
end
local fn, err = (loadstring or load)(src, "=Fust")
if not fn then
        warn("[ПП 0.44] Компиляция: " .. tostring(err))
        return
end
local ok, rerr = pcall(fn)
if not ok then
        warn("[ПП 0.44] Выполнение: " .. tostring(rerr))
        warn("[ПП 0.44] Скинь СКРИН консоли -- там будет строка [FUST] с местом падения, либо красное окно ошибки на экране.")
end
