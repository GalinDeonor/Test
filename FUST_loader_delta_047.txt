-- ЛОАДЕР «Пьяный Петух» 0.47 DELTA — ANDROID (БЕЗ ФРИКАМА)
-- Скачивает 6 частей (Fust1..Fust6) с GitHub, склеивает и запускает.
-- ПЕРЕД ЗАПУСКОМ: залей Fust1..Fust6, manifest.json и этот лоадер в КОРЕНЬ
-- репозитория GalinDeonor/Test (Add file -> Upload files -> Commit),
-- старые Fust1..6 заменятся, СТАРЫЕ ЛОАДЕРЫ (046, 045, 044, 037) УДАЛИ.
-- 0.47 DELTA = фрикам ВЫРЕЗАН (персонаж больше не замораживается),
-- авто-стрельба починена для телефона, остальное как в 0.39-линии.
-- Для XENO (ПК) есть отдельный файл FUST_xeno_047.lua — вставить целиком.
print("[ПП 0.47] Лоадер запущен, качаю 6 частей с GitHub...")
local B = "https://raw.githubusercontent.com/GalinDeonor/Test/refs/heads/main/Fust"
local want = {66398, 66416, 66408, 66419, 66401, 66338}
local TOTAL = 398380
local src = ""
for i = 1, 6 do
        local r
        local u = B .. i .. "?v=47"
        local ok, rr = pcall(game.HttpGet, game, u)
        if not ok or type(rr) ~= "string" or #rr < 1000 then
                local rf = request or http_request
                if rf then
                        local ok2, rr2 = pcall(rf, {Url = u, Method = "GET"})
                        if ok2 and type(rr2) == "table" then rr = rr2.Body or rr2.body end
                end
        end
        if type(rr) == "string" and #rr >= want[i] then
                r = rr
                print("[ПП 0.47] Часть " .. i .. ": " .. #rr .. " байт")
        else
                warn("[ПП 0.47] Часть " .. i .. " битая или не залита (нужно " .. want[i] .. " байт). Перезалей Fust" .. i .. " на GitHub.")
                return
        end
        src = src .. r
end
print("[ПП 0.47] Всего: " .. #src .. " байт")
if #src < TOTAL then
        warn("[ПП 0.47] Сумма меньше " .. TOTAL .. " -- какая-то часть битая, скинь скрин консоли.")
        return
end
local fn, err = (loadstring or load)(src, "=Fust")
if not fn then
        warn("[ПП 0.47] Компиляция: " .. tostring(err))
        return
end
local ok, rerr = pcall(fn)
if not ok then
        warn("[ПП 0.47] Выполнение: " .. tostring(rerr))
        warn("[ПП 0.47] Скинь СКРИН консоли -- там будет строка [FUST] с местом падения.")
end
