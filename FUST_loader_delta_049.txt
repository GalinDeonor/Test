-- ЛОАДЕР «Пьяный Петух» 0.49 DELTA — ANDROID (БЕЗ ФРИКАМА)
-- Скачивает 6 частей (Fust1..Fust6) с GitHub, склеивает и запускает.
-- ПЕРЕД ЗАПУСКОМ: залей Fust1..Fust6, manifest.json и этот лоадер в КОРЕНЬ
-- репозитория GalinDeonor/Test (Add file -> Upload files -> Commit) —
-- ВСЕ 7 файлов ОДНИМ КОММИТОМ. СТАРЫЕ ЛОАДЕРЫ (048, 047, 046, 045, 044, 037) УДАЛИ.
-- 0.49 = СТАБИЛЬНАЯ 0.39 + ТОЛЬКО авто-стрельба (WallCheck, телефон-фикс).
-- Фрикам, есп лута (новый), чамсы мебели — ВЫРЕЗАНЫ.
-- Для XENO (ПК) есть отдельный файл FUST_xeno_049.lua — вставить целиком.
print("[ПП 0.49] Лоадер запущен, качаю 6 частей с GitHub...")
local B = "https://raw.githubusercontent.com/GalinDeonor/Test/refs/heads/main/Fust"
local want = {62697, 62682, 62701, 62680, 62695, 62619}
local TOTAL = 376074
local src = ""
for i = 1, 6 do
        local r
        local u = B .. i .. "?v=49"
        local ok, rr = pcall(game.HttpGet, game, u)
        if not ok or type(rr) ~= "string" or #rr < 1000 then
                local rf = request or http_request
                if rf then
                        local ok2, rr2 = pcall(rf, {Url = u, Method = "GET"})
                        if ok2 and type(rr2) == "table" then rr = rr2.Body or rr2.body end
                end
        end
        if type(rr) == "string" and #rr >= want[i] then
                r = rr
                print("[ПП 0.49] Часть " .. i .. ": " .. #rr .. " байт")
        else
                warn("[ПП 0.49] Часть " .. i .. " битая или не залита (нужно " .. want[i] .. " байт). Перезалей Fust" .. i .. " на GitHub.")
                return
        end
        src = src .. r
end
print("[ПП 0.49] Всего: " .. #src .. " байт")
if #src < TOTAL then
        warn("[ПП 0.49] Сумма меньше " .. TOTAL .. " -- какая-то часть битая, скинь скрин консоли.")
        return
end
local fn, err = (loadstring or load)(src, "=Fust")
if not fn then
        warn("[ПП 0.49] Компиляция: " .. tostring(err))
        return
end
local ok, rerr = pcall(fn)
if not ok then
        warn("[ПП 0.49] Выполнение: " .. tostring(rerr))
        warn("[ПП 0.49] Скинь СКРИН консоли -- там будет строка [FUST] с местом падения.")
end
