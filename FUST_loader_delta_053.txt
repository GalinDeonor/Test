-- ЛОАДЕР «Пьяный Петух» 0.53 DELTA — ANDROID
-- Скачивает 6 частей (Fust1..Fust6) с GitHub, склеивает и запускает.
-- ПЕРЕД ЗАПУСКОМ: залей в КОРЕНЬ репозитория GalinDeonor/Test ОДНИМ КОММИТОМ:
--   Fust1..Fust6, manifest.json, FUST_loader_delta_053.txt, BUGS.md
-- СТАРЫЕ ЛОАДЕРЫ (052, 051, 050, 049...) УДАЛИ. Файл BUGS.md НЕ переименовывай NEVER.
-- 0.53 = возврат вырезанного (чамсы мебели, страница Логи) + прицел и пули
-- VOMAGLA + подсказки (i) + бинды-кнопки + фиксы лагов лута и чамсов лута.
-- Внутри все фиксы 0.52 (E-15/E-16/E-17). XENO (ПК): файл FUST_xeno_053.lua целиком.
print("[ПП 0.53] Лоадер запущен, качаю 6 частей с GitHub...")
local B = "https://raw.githubusercontent.com/GalinDeonor/Test/refs/heads/main/Fust"
local want = {77129, 77135, 77134, 77128, 77135, 77158}
local TOTAL = 462819
local src = ""
for i = 1, 6 do
        local r
        local u = B .. i .. "?v=53"
        local ok, rr = pcall(game.HttpGet, game, u)
        if not ok or type(rr) ~= "string" or #rr < 1000 then
                local rf = request or http_request
                if rf then
                        local ok2, rr2 = pcall(rf, {Url = u, Method = "GET"})
                        if ok2 and type(rr2) == "table" then rr = rr2.Body or rr2.body end
                end
        end
        if type(rr) == "string" and #rr >= want[i] then
                r = rr
                print("[ПП 0.53] Часть " .. i .. ": " .. #rr .. " байт")
        else
                warn("[ПП 0.53] Часть " .. i .. " битая или не залита (нужно " .. want[i] .. " байт). Перезалей Fust" .. i .. " на GitHub.")
                return
        end
        src = src .. r
end
print("[ПП 0.53] Всего: " .. #src .. " байт")
if #src < TOTAL then
        warn("[ПП 0.53] Сумма меньше " .. TOTAL .. " -- какая-то часть битая, скинь скрин консоли.")
        return
end
local fn, err = (loadstring or load)(src, "=Fust")
if not fn then
        warn("[ПП 0.53] Компиляция: " .. tostring(err))
        return
end
local ok, rerr = pcall(fn)
if not ok then
        warn("[ПП 0.53] Выполнение: " .. tostring(rerr))
        warn("[ПП 0.53] Скинь СКРИН консоли -- там будет строка [FUST] с местом падения.")
end
