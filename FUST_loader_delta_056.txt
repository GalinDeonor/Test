-- ЛОАДЕР «Пьяный Петух» 0.56 DELTA — ANDROID
-- Скачивает 6 частей (Fust1..Fust6) с GitHub, склеивает и запускает.
-- ПЕРЕД ЗАПУСКОМ: залей в КОРЕНЬ репозитория GalinDeonor/Test ОДНИМ КОММИТОМ:
--   Fust1..Fust6, manifest.json, FUST_loader_delta_056.txt, BUGS.md
-- СТАРЫЕ ЛОАДЕРЫ (055, 054, 053...) УДАЛИ. Файл BUGS.md НЕ переименовывай NEVER.
-- 0.56 = СТАРАЯ АВТО-СТРЕЛЬБА (0.39-линия, работает на Xeno; VOMAGLA не
-- стрелял) + ФРИКАМ БЕЗ КАСАНИЯ ПЕРСОНАЖА (разбор бана WALKSPEED 1000:
-- якорь-заморозка удалена, чит не пишет ни одного свойства персонажа)
-- + FOV-КРУГ «крутой» (вращающийся градиент: красно-чёрный / зелёный,
-- цвета и толщина настраиваются).
-- Внутри всё 0.55 (меню фрикама, логи, журнал, конфиги, X-Ray).
-- XENO (ПК): файл FUST_xeno_056.lua целиком.
print("[ПП 0.56] Лоадер запущен, качаю 6 частей с GitHub...")
local B = "https://raw.githubusercontent.com/GalinDeonor/Test/refs/heads/main/Fust"
local want = {83075, 83124, 83037, 83117, 83102, 83328}
local TOTAL = 498783
local src = ""
for i = 1, 6 do
        local r
        local u = B .. i .. "?v=56"
        local ok, rr = pcall(game.HttpGet, game, u)
        if not ok or type(rr) ~= "string" or #rr < 1000 then
                local rf = request or http_request
                if rf then
                        local ok2, rr2 = pcall(rf, {Url = u, Method = "GET"})
                        if ok2 and type(rr2) == "table" then rr = rr2.Body or rr2.body end
                end
        end
        if type(rr) == "string" and #rr >= want[i] then
                r = rr
                print("[ПП 0.56] Часть " .. i .. ": " .. #rr .. " байт")
        else
                warn("[ПП 0.56] Часть " .. i .. " битая или не залита (нужно " .. want[i] .. " байт). Перезалей Fust" .. i .. " на GitHub.")
                return
        end
        src = src .. r
end
print("[ПП 0.56] Всего: " .. #src .. " байт")
if #src < TOTAL then
        warn("[ПП 0.56] Сумма меньше " .. TOTAL .. " -- какая-то часть битая, скинь скрин консоли.")
        return
end
local fn, err = (loadstring or load)(src, "=Fust")
if not fn then
        warn("[ПП 0.56] Компиляция: " .. tostring(err))
        return
end
local ok, rerr = pcall(fn)
if not ok then
        warn("[ПП 0.56] Выполнение: " .. tostring(rerr))
        warn("[ПП 0.56] Скинь СКРИН консоли -- там будет строка [FUST] с местом падения.")
end
