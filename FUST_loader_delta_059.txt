-- ЛОАДЕР «Пьяный Петух» 0.59 DELTA — ANDROID
-- Скачивает 6 частей (Fust1..Fust6) с GitHub, склеивает и запускает.
-- ПЕРЕД ЗАПУСКОМ: залей в КОРЕНЬ репозитория GalinDeonor/Test ОДНИМ КОММИТОМ:
--   Fust1..Fust6, manifest.json, BUGS.md, FUST_loader_delta_059.txt
-- ЛОАДЕР FUST_loader_universal.txt НЕ МЕНЯЕТСЯ: сам качает 0.59 через manifest.
-- 0.59 = МУЛЬТИ-ТОЧКИ ФРИКАМА: SAVE POS добавляет точку (до 20), < > —
-- листать, TP CAM TO POS — лететь в выбранную, DEL POS — удалить.
-- Точки хранятся в FUST_freecam_pos.json и не пропадают после перезахода.
-- Внутри всё 0.58 (X-Ray ловит стены E-23, фрикам E-22, стрельба, FOV-круг).
print("[ПП 0.59] Лоадер запущен, качаю 6 частей с GitHub...")
local B = "https://raw.githubusercontent.com/GalinDeonor/Test/refs/heads/main/Fust"
local want = {85923, 85908, 85603, 85898, 85947, 86456}
local TOTAL = 515735
local src = ""
for i = 1, 6 do
        local r
        local u = B .. i .. "?v=59"
        local ok, rr = pcall(game.HttpGet, game, u)
        if not ok or type(rr) ~= "string" or #rr < 1000 then
                local rf = request or http_request
                if rf then
                        local ok2, rr2 = pcall(rf, {Url = u, Method = "GET"})
                        if ok2 and type(rr2) == "table" then rr = rr2.Body or rr2.body end
                end
        end
        if type(rr) == "string" and #rr >= want[i] then
                r = rr
                print("[ПП 0.59] Часть " .. i .. ": " .. #rr .. " байт")
        else
                warn("[ПП 0.59] Часть " .. i .. " битая или не залита (нужно " .. want[i] .. " байт). Перезалей Fust" .. i .. " на GitHub.")
                return
        end
        src = src .. r
end
print("[ПП 0.59] Всего: " .. #src .. " байт")
if #src < TOTAL then
        warn("[ПП 0.59] Сумма меньше " .. TOTAL .. " -- какая-то часть битая, скинь скрин консоли.")
        return
end
local fn, err = (loadstring or load)(src, "=Fust")
if not fn then
        warn("[ПП 0.59] Компиляция: " .. tostring(err))
        return
end
local ok, rerr = pcall(fn)
if not ok then
        warn("[ПП 0.59] Выполнение: " .. tostring(rerr))
        warn("[ПП 0.59] Скинь СКРИН консоли -- там будет строка [FUST] с местом падения.")
end
