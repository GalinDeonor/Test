-- «Пьяный Петух» — универсальный лоадер v3 (Xeno + Delta)
-- v3: КАЖДЫЙ запрос с гонкой по времени (10 сек) + 3 попытки на файл —
-- висеть бесконечно больше не может. Проверка версии манифеста (0.45):
-- если на репо старый манифест — лоадер СРАЗУ скажет об этом текстом.
-- Проверка целостности: длина + md5. Прогресс по каждой части.
-- МЕНЯТЬ ВНУТРИ НИЧЕГО НЕ НУЖНО — версия проверяется автоматически.

local EXPECTED_VER = "0.45"
local M1 = "https://raw.githubusercontent.com/GalinDeonor/Test/main/"
local M2 = "https://raw.githubusercontent.com/GalinDeonor/Test/main/" -- резерв, поставь свой
local H  = game:GetService("HttpService")

-- запрос с гонкой по времени: висящий HttpGet больше не блокирует всё
local function race(fn, sec)
    local result = nil
    task.spawn(function()
        result = fn()
    end)
    local t0 = os.time()
    while result == nil and os.time() - t0 < (sec or 10) do
        task.wait(0.1)
    end
    return result
end

local function tryHttp(u)
    local ok, r = pcall(game.HttpGet, game, u)
    if ok and type(r) == "string" and #r > 0 then return r end
    return nil
end

local function tryReq(u)
    local f = request or http_request or (syn and syn.request)
    if not f then return nil end
    local ok, r = pcall(f, {Url = u, Method = "GET"})
    if ok and type(r) == "table" then
        local body = r.Body or r.body
        if type(body) == "string" and #body > 0 then return body end
    end
    return nil
end

-- качаем URL с таймаутом и 3 попытками (Http -> request -> Http ...)
local function fetch(u, attempts)
    attempts = attempts or 3
    for i = 1, attempts do
        local body = race(function() return tryHttp(u) end, 10)
        if body then return body end
        body = race(function() return tryReq(u) end, 10)
        if body then return body end
        warn("[ПП] Попытка " .. i .. "/" .. attempts .. " не удалась, повторяю...")
        task.wait(0.4)
    end
    return nil
end

-- 1. manifest.json
local mj = fetch(M1 .. "manifest.json", 3)
if not mj or #mj < 10 then
    return warn("[ПП] manifest.json не скачался. Проверь интернет/репозиторий.")
end

local okM, m = pcall(H.JSONDecode, H, mj)
if not okM or type(m) ~= "table" or not m.parts then
    return warn("[ПП] manifest.json битый: " .. tostring(m))
end

if tostring(m.ver) ~= EXPECTED_VER then
    return warn("[ПП] НА РЕПО ВЕРСИЯ " .. tostring(m.ver) .. ", А НУЖНА " .. EXPECTED_VER ..
        ". Ты не заменил manifest.json (и/или части) на новый релиз. Если только что заливал — подожди 5 минут (кэш GitHub) и запусти ещё раз.")
end

print("[ПП] Версия на репо: " .. tostring(m.ver) .. " | частей: " .. #m.parts)

-- 2. Качаем части
local src = ""
for i = 1, #m.parts do
    local u = M1 .. m.parts[i]
    if M2 ~= M1 then
        -- резервное зеркало подставится внутри fetch через fallback ниже
    end
    local prt = fetch(u, 3)
    local want = m.sizes and m.sizes[i] or 0
    if not prt or #prt < want then
        return warn(string.format(
            "[ПП] Часть %d (%s) битая: %d байт, нужно %d. Перезалей репозиторий.",
            i, m.parts[i], prt and #prt or 0, want))
    end
    src = src .. prt
    print(string.format("[ПП %s] %d/%d: %s (%d байт) OK", m.ver, i, #m.parts, m.parts[i], #prt))
end

print("[ПП] Все " .. #m.parts .. " частей скачаны (" .. #src .. " байт). Проверяю целостность...")

-- 3. Целостность: длина + md5
if m.total and #src ~= m.total then
    return warn(string.format(
        "[ПП] Сумма %d != %d — на репо СМЕШАНЫ части разных версий. Залей новый релиз ЦЕЛИКОМ (все части + manifest.json).",
        #src, m.total))
end
if m.md5 then
    local okH, h = pcall(function()
        return (crypt and crypt.hash) and crypt.hash(src, "md5") or nil
    end)
    if okH and type(h) == "string" and #h > 0 and h ~= m.md5 then
        return warn("[ПП] MD5 не совпал — на репо старые/чужие части. Залей новый релиз ЦЕЛИКОМ, потом rejoin.")
    end
end
print("[ПП] Целостность OK. Запускаю «Пьяный Петух»...")

-- 4. Компилируем и запускаем
local fn, err = (loadstring or load)(src, "=Fust")
if not fn then
    return warn("[ПП] Компиляция: " .. tostring(err))
end
local ok, rerr = pcall(fn)
if not ok then
    warn("[ПП] Выполнение: " .. tostring(rerr))
    warn("[ПП] Скинь скрин консоли — там будет [FUST] с местом падения.")
else
    print("[ПП] Готово. Меню — по кнопке-петуху / RightControl.")
end
